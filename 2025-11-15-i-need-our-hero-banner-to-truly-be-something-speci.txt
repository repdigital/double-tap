     Purpose: Physics-based kinetic typography with architectural fulcrum

     'use client'

     import { useEffect, useState } from 'react'
     import { useSpring, animated } from '@react-spring/web'
     import { HERO_CONSTANTS } from '@/lib/hero-constants'

     interface BalancedHeadlineProps {
       topLine: string
       bottomLine: string
       enableBreathing?: boolean
       enableHover?: boolean
     }

     export function BalancedHeadline({ 
       topLine, 
       bottomLine,
       enableBreathing = true,
       enableHover = false 
     }: BalancedHeadlineProps) {
       const [isHovered, setIsHovered] = useState(false)
       
       // Main tilt spring
       const [tiltProps, tiltApi] = useSpring(() => ({
         rotateZ: HERO_CONSTANTS.BALANCE_PHYSICS.entry.from,
         config: HERO_CONSTANTS.BALANCE_PHYSICS.entry.config,
       }))
       
       // Entry animation on mount
       useEffect(() => {
         const timer = setTimeout(() => {
           tiltApi.start({ 
             rotateZ: HERO_CONSTANTS.BALANCE_PHYSICS.entry.to 
           })
         }, HERO_CONSTANTS.ANIMATION.headline.delay)
         
         return () => clearTimeout(timer)
       }, [tiltApi])
       
       // Breathing cycle
       useEffect(() => {
         if (!enableBreathing) return
         
         const breathe = () => {
           const { amplitude, duration } = 
     HERO_CONSTANTS.BALANCE_PHYSICS.breathing
           
           tiltApi.start({
             to: async (next) => {
               await next({ rotateZ: amplitude, config: { duration: duration
      / 3 } })
               await next({ rotateZ: -amplitude, config: { duration: 
     duration / 3 } })
               await next({ rotateZ: 0, config: { duration: duration / 3 } 
     })
             }
           })
         }
         
         const interval = setInterval(breathe, 
     HERO_CONSTANTS.BALANCE_PHYSICS.breathing.interval)
         return () => clearInterval(interval)
       }, [enableBreathing, tiltApi])
       
       // Hover interaction
       useEffect(() => {
         if (!enableHover) return
         
         const { tilt, config } = HERO_CONSTANTS.BALANCE_PHYSICS.hover
         
         if (isHovered) {
           tiltApi.start({ rotateZ: tilt, config })
         } else {
           tiltApi.start({ rotateZ: 0, config })
         }
       }, [isHovered, enableHover, tiltApi])
       
       return (
         <div 
           className="relative"
           onMouseEnter={() => enableHover && setIsHovered(true)}
           onMouseLeave={() => enableHover && setIsHovered(false)}
         >
           {/* Architectural Fulcrum */}
           <div 
             className="absolute left-1/2 top-1/2 -translate-x-1/2 
     -translate-y-1/2 w-0.5 h-10 bg-[var(--hero-accent)] opacity-40 z-10"
             style={{
               boxShadow: '0 0 20px var(--hero-accent-glow)'
             }}
           />
           
           {/* Balanced Typography */}
           <animated.div 
             style={{ 
               rotateZ: tiltProps.rotateZ.to(r => `${r}deg`),
               transformOrigin: 'center center'
             }}
             className="relative z-0"
           >
             <h1 className="text-center">
               <span className="block hero-headline mb-2">
                 {topLine}
               </span>
               <span className="block hero-headline">
                 {bottomLine}
               </span>
             </h1>
           </animated.div>
         </div>
       )
     }

     'use client'

     import { ReactNode } from 'react'
     import { motion, useScroll, useTransform } from 'framer-motion'
     import { useHeroLayoutState } from '@/lib/hooks/useHeroLayoutState'
     import { HERO_CONSTANTS } from '@/lib/hero-constants'

     interface DynamicHeroLayoutProps {
       leftContent: ReactNode
       rightContent: ReactNode
       centerContent: ReactNode
     }

     export function DynamicHeroLayout({ 
       leftContent, 
       rightContent,
       centerContent 
     }: DynamicHeroLayoutProps) {
       
       const { scrollYProgress } = useScroll()
       const layoutState = useHeroLayoutState(scrollYProgress)
       
       // Interpolate grid columns based on scroll
       const gridTemplateColumns = useTransform(
         scrollYProgress,
         [0, HERO_CONSTANTS.LAYOUT_TRANSFORM.triggerPoint],
         ['1fr', `${HERO_CONSTANTS.SPLIT_LEFT}fr 
     ${HERO_CONSTANTS.SPLIT_RIGHT}fr`]
       )
       
       // Interpolate opacity for centered vs split content
       const centerOpacity = useTransform(
         scrollYProgress,
         [0, HERO_CONSTANTS.LAYOUT_TRANSFORM.triggerPoint - 0.1, 
     HERO_CONSTANTS.LAYOUT_TRANSFORM.triggerPoint],
         [1, 1, 0]
       )
       
       const splitOpacity = useTransform(
         scrollYProgress,
         [0, HERO_CONSTANTS.LAYOUT_TRANSFORM.triggerPoint, 
     HERO_CONSTANTS.LAYOUT_TRANSFORM.triggerPoint + 0.1],
         [0, 0, 1]
       )
       
       return (
         <div className="relative w-full h-screen overflow-hidden">
           {/* Centered Layout (visible at top) */}
           <motion.div
             className="absolute inset-0 flex items-center justify-center"
             style={{ opacity: centerOpacity }}
           >
             <div className="max-w-4xl mx-auto px-6">
               {centerContent}
             </div>
           </motion.div>
           
           {/* Split Layout (visible after scroll) */}
           <motion.div
             className="absolute inset-0 grid items-center gap-8 px-6 
     lg:px-12 max-w-7xl mx-auto"
             style={{ 
               gridTemplateColumns,
               opacity: splitOpacity 
             }}
           >
             <div className="flex flex-col justify-center space-y-8">
               {leftContent}
             </div>
             <div className="flex items-center justify-center">
               {rightContent}
             </div>
           </motion.div>
         </div>
       )
     }

     Hook Implementation (lib/hooks/useHeroLayoutState.ts):
     'use client'

     import { useEffect, useState } from 'react'
     import { MotionValue } from 'framer-motion'
     import { HERO_CONSTANTS } from '@/lib/hero-constants'

     export type LayoutState = 'centered' | 'transitioning' | 'split'

     export function useHeroLayoutState(scrollProgress: 
     MotionValue<number>): LayoutState {
       const [state, setState] = useState<LayoutState>('centered')
       
       useEffect(() => {
         return scrollProgress.on('change', (latest) => {
           const trigger = HERO_CONSTANTS.LAYOUT_TRANSFORM.triggerPoint
           
           if (latest < trigger - 0.1) {
             setState('centered')
           } else if (latest >= trigger - 0.1 && latest < trigger + 0.1) {
             setState('transitioning')
           } else {
             setState('split')
           }
         })
       }, [scrollProgress])
       
       return state
     }

         if (!ctx) return
         
         const resize = () => {
           const dpr = window.devicePixelRatio || 1
           const rect = canvas.getBoundingClientRect()
           
           canvas.width = rect.width * dpr
           canvas.height = rect.height * dpr
           
           ctx.scale(dpr, dpr)
           
           // Generate noise
           const imageData = ctx.createImageData(canvas.width, 
     canvas.height)
           const data = imageData.data
           
           for (let i = 0; i < data.length; i += 4) {
             const noise = Math.random() * 255
             data[i] = noise     // R
             data[i + 1] = noise // G
             data[i + 2] = noise // B
             data[i + 3] = opacity * 255 // A
           }
           
           ctx.putImageData(imageData, 0, 0)
         }
         
         resize()
         window.addEventListener('resize', resize)
         
         return () => window.removeEventListener('resize', resize)
       }, [opacity])
       
       return (
         <>
           {/* Noise Canvas */}
           <canvas 
             ref={canvasRef}
             className="absolute inset-0 pointer-events-none"
             style={{ mixBlendMode: 'overlay' }}
           />
           
           {/* Optional Grid Overlay */}
           {gridEnabled && (
             <div 
               className="absolute inset-0 pointer-events-none"
               style={{
                 backgroundImage: `
                   linear-gradient(var(--hero-border) 1px, transparent 1px),
                   linear-gradient(90deg, var(--hero-border) 1px, 
     transparent 1px)
                 `,
                 backgroundSize: '40px 40px',
                 opacity: 0.08
               }}
             />
           )}

     }
